for(int i=0;i<config.Tracks.Length;i++){
 var track=config.Tracks[i];var rect=(UnityEngine.RectTransform)targets.GetArrayElementAtIndex(i).objectReferenceValue;
 rows.Add(new {name=track.Part,x=rect.anchoredPosition.x,y=rect.anchoredPosition.y,scale=track.Scale,offsetX=track.Offset.x,offsetY=track.Offset.y,delay=track.Delay,duration=track.Duration,sibling=rect.GetSiblingIndex()});
}
return new {playing=UnityEditor.EditorApplication.isPlaying,scene=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path,serialized=UnityEditor.EditorJsonUtility.ToJson(config),tracks=rows};
