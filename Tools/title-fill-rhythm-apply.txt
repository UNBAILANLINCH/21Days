  var fill=(UnityEngine.RectTransform)paper.Find("底部渐变");

   track.FindPropertyRelative("curve").animationCurveValue=new UnityEngine.AnimationCurve(
    new UnityEngine.Keyframe(0f,0f,0f,0f),new UnityEngine.Keyframe(0.18f,0.03f,0.5f,0.5f),

  }
