```mermaid
flowchart LR
  xs(["现生"]) --> jz(["镜子"])

  g2a -->|"同盟？"| tlB["同类怪"]
  tlB --> jd["节点"]
  zd -->|"战斗（正面？）"| g2b["怪2"]
  g2b -->|"潜行刺杀"| tlC["同类怪"]
  tlC --> jd
  jd --> boss["boss战（机制？）"]
  lsj <--> boss
```
